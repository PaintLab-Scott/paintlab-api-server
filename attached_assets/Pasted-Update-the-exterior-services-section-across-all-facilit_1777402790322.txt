PART 1 — EXTERIOR WASH PRICING LOGIC (REQUIRED)

Ensure exterior wash services are calculated using tier-based visit frequency.

Use this function:

function calculateWashMonthly(zones, tier) {
  const visits = VISITS_PER_YEAR[tier];
  if (!visits) return 0;

  let total = 0;

  zones.forEach(zone => {
    if (WASH_ZONES[zone]) {
      total += WASH_ZONES[zone];
    }
  });

  return (total * visits) / 12;
}
